#include <stdio.h>
#include <stdlib.h>
void scanm(int a[10][10],int r, int c)
{
    for(int i=0;i<r;i++)
    {
        for(int j=0;j<c;j++)
        {
            scanf("%d",&a[i][j]);
        }
    }
    
}

void comp(int a[10][10],int b[20][3],int r, int c)
{ int k=1;
    b[0][0]=r; b[0][1]=c;
    for(int i=0;i<r;i++)
    {
        for(int j=0;j<c;j++)
        {
            if(a[i][j]!=0)
            {
                b[k][0]=i;
                b[k][1]=j;
                b[k][2]=a[i][j];
                k=k+1;
            }
        }
       
    }
     b[0][2]=k-1;
        
}
void dispcomp(int b[20][3],int r)
{
    for (int i=0;i<r;i++)
    {
        for(int j=0;j<3;j++)
        {
            printf("%d ",b[i][j]);
        }
        printf("\n");
    }
}

void trans(int b[20][3],int t[20][3],int r,int c)
{int k=1; t[0][0]=c; t[0][1]=r;t[0][2]=b[0][2];
    for(int j=0;j<c;j++)
    {
        for(int i=1;i<=b[0][2];i++)
        {
            if(b[i][1]==j)
            {
                t[k][0]=b[i][1];
                t[k][1]=b[i][0];
                t[k][2]=b[i][2];k++;
            }
        }
    }
}

void ftrans(int a[20][3], int t[20][3], int r, int c)
{
     int x;int k;int j;int i;
    x = k = 0;
    t[0][0] = a[0][1];
    t[0][1] = a[0][0];
    t[0][2] = a[0][2];
    int total = a[0][2];
    
    int count[10] = {0};
    int stridx[10] = {0};

    for(j=1;j <= total;j++)
        {
            count[a[j][1]]++;
        }

    stridx[0] = 1;
    for(i=1;i < c;i++)
    {
        stridx[i] =stridx[i - 1] + count[i - 1];
    }

    for(i=1;i<=total;i++)
    {
        x =a[i][1];
        k =stridx[x];
        t[k][0]=a[i][1];
        t[k][1]=a[i][0];
        t[k][2]=a[i][2];
        stridx[x]++;
    }
}


int main() {
   int a[10][10];int b[20][3];int r; int c;int t[20][3];int ft[20][3];int choice;
   printf("give total rows and columns:\n");
   scanf("%d %d",&r,&c);
   printf("ENTER ELEMENTS:\n");
   scanm(a,r,c);
   comp(a,b,r,c);
   printf("compact form is: \n");
   dispcomp(b,b[0][2]+1);
  
   
   while(1)
   {
       printf("give your choice: \n");
       printf("1.simple tranpose\n");
       printf("2.Fast tranpose\n");
       printf("3.exit\n");
       scanf("%d",&choice);
       
switch(choice) 
{
    
case 1:
   printf("simple tranpose is: \n");
    trans(b,t,r,c);
   dispcomp(t,b[0][2]+1);
   break;
   
   case 2:
   printf("fast tranpose is:  \n");
   ftrans(b,ft,r,c);

   case 3:

}
}
