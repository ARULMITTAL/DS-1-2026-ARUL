#include <stdio.h>
#include <string.h>
#include <stdlib.h>

typedef struct hospital
{
    char patname[20];
    int patid;
    float patbill;
} hospital;

void accept(hospital patient[], int size)
{
    int i;

    for(i = 0; i < size; i++)
    {
        printf("\nPatient %d\n", i + 1);
        printf("Patient name: ");
        scanf("%s",patient[i].patname);

        printf("Patient ID: ");
        scanf("%d",&patient[i].patid);

        printf("Patient bill: ");
        scanf("%f",&patient[i].patbill);
    }
}

void display(hospital patient[], int size)
{
    int i;

    printf(" Patient Details \n");
    printf("name\tid\tbill\n");
    for(i = 0; i < size; i++)
    {
       
        printf("%s\t%d\t%.2f\n",patient[i].patname,patient[i].patid,patient[i].patbill);
       
    }
}

void bubblesort(hospital patient[],int m)
{
int i,j;
struct hospital temp;
for(i=0;i<m-1;i++)
{
	for(j=0;j<m-1;j++)
	{
	if(patient[j].patid>patient[j+1].patid)
		{
		temp=patient[j];
		patient[j]=patient[j+1];
		patient[j+1]=temp;
		}
	}
}
}

void insertionsort(hospital patient[], int m)
{
    int i, j;
    struct hospital temp;
    for(i = 1; i < m; i++)
    {
        temp = patient[i];
        j = i - 1;

        while(j >= 0 && patient[j].patid > temp.patid)
        {
            patient[j + 1] = patient[j];
            j--;
        }
        patient[j + 1] = temp;
    }
}

void selectionsort(hospital patient[], int m)
{
    int i, j, min_idx;
    struct hospital temp;
    for(i = 0; i < m - 1; i++)
    {
        min_idx = i;
        for(j = i + 1; j < m; j++)
        {
            if(patient[j].patid < patient[min_idx].patid)
            {
                min_idx = j;
            }
        }
        if(min_idx != i)
        {
            temp = patient[i];
            patient[i] = patient[min_idx];
            patient[min_idx] = temp;
        }
    }
}

void shellsort(hospital patient[], int m)
{
    int i, j, gap;
    struct hospital temp;
    for(gap = m / 2; gap > 0; gap /= 2)
    {
        for(i = gap; i < m; i++)
        {
            temp = patient[i];
            for(j = i; j >= gap && patient[j - gap].patid > temp.patid; j -= gap)
            {
                patient[j] = patient[j - gap];
            }
            patient[j] = temp;
        }
    }
}

int binarySearch(hospital patient[], int m, int key)
{
    int low = 0, high = m - 1, mid;
    while (low <= high)
    {
        mid = low + (high - low) / 2;
        if (patient[mid].patid == key)
        {
            printf("%d is found at position %d using Binary Search\n", key, mid + 1);
            return 1;
        }
        if (patient[mid].patid < key)
            low = mid + 1;
        else
            high = mid - 1;
    }
    return 0;
}

int linear(hospital patient[], int m, char searchName[])
{
    int i, found = 0;
    for(i = 0; i < m; i++)
    {
        if(strcmp(patient[i].patname, searchName) == 0)
        {
            printf("Patient Name %s found at position %d (ID: %d, Bill: %.2f)\n", 
                   searchName, i + 1, patient[i].patid, patient[i].patbill);
            found = 1;
        }
    }
    if(found)
        return 1;
    return 0;
}
